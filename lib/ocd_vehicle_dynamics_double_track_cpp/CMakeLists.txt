cmake_minimum_required(VERSION 3.12)
set(PROJECT_NAME ocd_vehicle_dynamics_double_track_cpp)
project(${PROJECT_NAME})

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

set(MODEL_DEPENDENCIES
  Eigen3
  ament_cmake
  tum_types_cpp
  tsl_logger_cpp
  tum_helpers_cpp
  ocd_types_cpp
  ocd_tire_model_base_cpp
  param_management_cpp
  ocd_aerodynamics_model_base_cpp
  ocd_vehicle_dynamics_model_base_cpp
)

foreach(pkg_name ${MODEL_DEPENDENCIES})
  find_package(${pkg_name} REQUIRED)
endforeach()

add_library(${PROJECT_NAME} INTERFACE)
target_include_directories(${PROJECT_NAME} 
    INTERFACE 
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
)
target_compile_features(${PROJECT_NAME} INTERFACE c_std_99 cxx_std_17)
target_compile_options(${PROJECT_NAME} INTERFACE -fopenmp) #-dndebug -fno-math-errno -fopenmp)

install(
  DIRECTORY include/
  DESTINATION include
)
install(
  TARGETS ${PROJECT_NAME}
  EXPORT export_${PROJECT_NAME}
  DESTINATION lib/${PROJECT_NAME}
  LIBRARY DESTINATION lib
  ARCHIVE DESTINATION lib
  RUNTIME DESTINATION bin
  INCLUDES DESTINATION include
)

ament_target_dependencies(${PROJECT_NAME} INTERFACE ${MODEL_DEPENDENCIES})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET) # so client can also call target_link_libraries(template_lib_cpp::template_lib_cpp)
ament_export_dependencies(${MODEL_DEPENDENCIES}) # other dependencies so client does not need to call find_package on them

if(BUILD_TESTING)
  enable_testing()
  add_executable(generic_radial_lookup_test test/generic_radial_lookup.cpp)
  target_include_directories(generic_radial_lookup_test PRIVATE include)
  target_compile_features(generic_radial_lookup_test PRIVATE cxx_std_17)
  add_test(NAME generic_radial_lookup COMMAND generic_radial_lookup_test)
  find_package(ocd_tire_models_cpp REQUIRED)
  find_package(ocd_aerodynamics_models_cpp REQUIRED)
  add_executable(suspension_geometry_test test/suspension_geometry.cpp)
  target_link_libraries(suspension_geometry_test ${PROJECT_NAME})
  ament_target_dependencies(suspension_geometry_test ocd_tire_models_cpp ocd_aerodynamics_models_cpp)
  target_compile_features(suspension_geometry_test PRIVATE cxx_std_20)
  add_test(NAME suspension_geometry COMMAND suspension_geometry_test)
  add_executable(mf62_vertical_test test/mf62_vertical.cpp)
  target_include_directories(mf62_vertical_test PRIVATE include)
  add_test(NAME mf62_vertical COMMAND mf62_vertical_test)
  add_executable(suspension_output_test test/suspension_output.cpp)
  target_link_libraries(suspension_output_test ${PROJECT_NAME})
  target_compile_features(suspension_output_test PRIVATE cxx_std_20)
  ament_target_dependencies(suspension_output_test ocd_tire_models_cpp ocd_aerodynamics_models_cpp)
  add_test(NAME suspension_output COMMAND suspension_output_test)
  foreach(test_name radial_root_contract mf_schwift_ring nonlinear_radial_contact tire_load_memory longitudinal_relaxation point_road_energy external_wheel_forces material_velocity coupled_geometry five_point_contact contact_energy contact_frame contact_energy_envelope belt_energy heave_energy heave_coil_energy damper_curves)
    add_executable(${test_name}_test test/${test_name}.cpp)
    target_link_libraries(${test_name}_test ${PROJECT_NAME})
    target_compile_features(${test_name}_test PRIVATE cxx_std_20)
    ament_target_dependencies(${test_name}_test ocd_tire_models_cpp ocd_aerodynamics_models_cpp)
    add_test(NAME ${test_name} COMMAND ${test_name}_test)
  endforeach()
endif()

ament_package()
